getHTFName(string htf) =>
    seconds = timeframe.in_seconds(htf)
    if seconds >= 2592000
        "1M"
    else if seconds == 604800
        "1W"
    else if seconds >= 86400
        str.tostring(math.round(seconds / 86400)) + "D"
    else if seconds == 5400
        "90M"
    else if seconds >= 3600
        str.tostring(math.round(seconds / 3600)) + "H"
    else if seconds >= 60
        str.tostring(math.round(seconds / 60)) + "M"
    else
        str.tostring(seconds) + "S"

getRemainingTimeHTF(string htf) =>
    var string r = ""
    if barstate.isrealtime
        timeRemaining = (time_close(htf) - timenow) / 1000
        if timeRemaining > 0
            hours   = math.floor(timeRemaining / 3600)
            minutes = math.floor((timeRemaining % 3600) / 60)
            seconds = math.floor(timeRemaining % 60)
            r := (hours > 0 ? str.tostring(hours, "00") + ":" : "") +
                 str.tostring(minutes, "00") + ":" +
                 str.tostring(seconds, "00")
        else
            r := "00:00:00"
    else
        r := "N/A"
    r

get_dynamic_time(string htf) =>
    tf_secs = timeframe.in_seconds(htf)
    res = ""
    if tf_secs >= 2592000
        res := str.format("{0,date,MMM}", time)
    else if tf_secs >= 604800
        res := str.upper(str.format("{0,date,dd MMM}", time))
    else if tf_secs >= 86400
        res := str.format("{0,date,EEE}", time)
    else
        res := str.format("{0,number,00}:{1,number,00}", hour(time), minute(time))
    res

method Reorder(CandleSet candleSet, int offset) =>
    for l in sweep_lines
        line.delete(l)
    array.clear(sweep_lines)

    size = candleSet.candles.size()
    float highest_point = 0.0

    if size > 0
        for i = size-1 to 0
            Candle candle = candleSet.candles.get(i)
            t_buffer = offset + ((settingsCandle.width + settingsCandle.buffer) * (size - i - 1))
            posX = bar_index + t_buffer
            midX = posX + (settingsCandle.width / 2)
            r_pos = posX + settingsCandle.width

            highest_point := math.max(highest_point, candle.h)

            box.set_left(candle.body, posX)
            box.set_right(candle.body, posX + settingsCandle.width)
            line.set_x1(candle.wick_up, midX), line.set_x2(candle.wick_up, midX)
            line.set_x1(candle.wick_down, midX), line.set_x2(candle.wick_down, midX)

            if i == 0
                candle.deleteProjections()

                if proj_o_enabled
                    candle.line_o := line.new(candle.o_idx, candle.o, midX, candle.o, style=getProjLineStyle(proj_o_style), color=proj_o_col, width=proj_o_width)
                    if proj_lbl_enabled
                        candle.lbl_o := label.new(r_pos, candle.o, str.tostring(candle.o, format.mintick), style=label.style_label_left, color=color.new(color.black, 100), textcolor=proj_lbl_col, size=getProjLabelSize(proj_lbl_size))

                if proj_hl_enabled
                    candle.line_h := line.new(candle.h_idx, candle.h, midX, candle.h, style=getProjLineStyle(proj_hl_style), color=proj_hl_col, width=proj_hl_width)
                    candle.line_l := line.new(candle.l_idx, candle.l, midX, candle.l, style=getProjLineStyle(proj_hl_style), color=proj_hl_col, width=proj_hl_width)
                    if proj_lbl_enabled
                        candle.lbl_h := label.new(r_pos, candle.h, str.tostring(candle.h, format.mintick), style=label.style_label_left, color=color.new(color.black, 100), textcolor=proj_lbl_col, size=getProjLabelSize(proj_lbl_size))
                        candle.lbl_l := label.new(r_pos, candle.l, str.tostring(candle.l, format.mintick), style=label.style_label_left, color=color.new(color.black, 100), textcolor=proj_lbl_col, size=getProjLabelSize(proj_lbl_size))

                if proj_c_enabled
                    candle.line_c := line.new(bar_index, candle.c, midX, candle.c, style=getProjLineStyle(proj_c_style), color=proj_c_col, width=proj_c_width)
                    if proj_lbl_enabled
                        candle.lbl_c := label.new(r_pos, candle.c, str.tostring(candle.c, format.mintick), style=label.style_label_left, color=color.new(color.white, 100), textcolor=proj_lbl_col, size=getProjLabelSize(proj_lbl_size))
            else
                candle.deleteProjections()

            if i < size - 1
                Candle prev = candleSet.candles.get(i+1)
                prevPosX = offset + ((settingsCandle.width + settingsCandle.buffer) * (size - (i+1) - 1))
                prevMidX = bar_index + prevPosX + (settingsCandle.width / 2)

                if show_sweeps
                    if candle.h > prev.h and candle.c < prev.h
                        sweep_lines.push(line.new(prevMidX, prev.h, posX + settingsCandle.width, prev.h, color=settingsCandle.line_color))
                    if candle.l < prev.l and candle.c > prev.l
                        sweep_lines.push(line.new(prevMidX, prev.l, posX + settingsCandle.width, prev.l, color=settingsCandle.line_color))

        if show_htf_info
            label.delete(candleSet.infoLabel)
            string labelTxt = getHTFName(candleSet.settings.htf) + "\n(" + getRemainingTimeHTF(candleSet.settings.htf) + ")"
            labelX = bar_index + offset + ((settingsCandle.width + settingsCandle.buffer) * (size - 1) / 2)

            candleSet.infoLabel := label.new(labelX, highest_point, labelTxt,
                                 color=color.new(color.white, 100),
                                 textcolor=info_col,
                                 style=label.style_label_down,
                                 size=info_size,
                                 text_font_family=font.family_monospace)

method Monitor(CandleSet candleSet) =>
    isNew = ta.change(time(candleSet.settings.htf)) != 0

    if not isNew
        candleSet.htf_high_trace := math.max(high, candleSet.htf_high_trace)
        candleSet.htf_low_trace  := math.min(low, candleSet.htf_low_trace)

    if isNew
        candleSet.htf_start_bar  := bar_index
        candleSet.htf_open_price := open
        candleSet.htf_high_trace := high
        candleSet.htf_low_trace  := low

        Candle candle = Candle.new(open, close, high, low, bar_index, bar_index, bar_index, bar_index)
        bull = candle.c > candle.o
        candle.body := box.new(bar_index, math.max(candle.o, candle.c), bar_index + 2, math.min(candle.o, candle.c),
             bull ? settingsCandle.bull_border : settingsCandle.bear_border, 1, bgcolor = bull ? settingsCandle.bull_body : settingsCandle.bear_body)
        candle.wick_up := line.new(bar_index, candle.h, bar_index, math.max(candle.o, candle.c), color=bull ? settingsCandle.bull_wick : settingsCandle.bear_wick)
        candle.wick_down := line.new(bar_index, math.min(candle.o, candle.c), bar_index, candle.l, color=bull ? settingsCandle.bull_wick : settingsCandle.bear_wick)
        candleSet.candles.unshift(candle)
        if candleSet.candles.size() > candleSet.settings.max_display
            Candle del = candleSet.candles.pop()
            box.delete(del.body), line.delete(del.wick_up), line.delete(del.wick_down)
            del.deleteProjections()
    candleSet

method Update(CandleSet candleSet, int offset) =>
    if candleSet.candles.size() > 0
        Candle candle = candleSet.candles.get(0)

        if high > candle.h
            candle.h := high
            candle.h_idx := bar_index
        if low < candle.l
            candle.l := low
            candle.l_idx := bar_index

        candle.c := close
        candle.c_idx := bar_index

        bull = candle.c > candle.o
        box.set_top(candle.body, math.max(candle.o, candle.c)), box.set_bottom(candle.body, math.min(candle.o, candle.c))
        box.set_bgcolor(candle.body, bull ? settingsCandle.bull_body : settingsCandle.bear_body)
        line.set_y1(candle.wick_up, candle.h), line.set_y2(candle.wick_up, math.max(candle.o, candle.c))
        line.set_y1(candle.wick_down, candle.l), line.set_y2(candle.wick_down, math.min(candle.o, candle.c))

        candleSet.Reorder(offset)
    candleSet

f_findLevelOrigin(isBull, lvl) =>
    int offset = 0
    for j = 1 to 400
        match = isBull ? (math.abs(low[j] - lvl) < syminfo.mintick) : (math.abs(high[j] - lvl) < syminfo.mintick)
        if match
            offset := j
            break
    offset

f_findOHLCOrigin(isBull, lvl, tf) =>
    htf_curr_start = ta.valuewhen(ta.change(time(tf)), bar_index, 0)
    htf_prev_start = ta.valuewhen(ta.change(time(tf)), bar_index, 1)
    htf_prev_end   = htf_curr_start - 1
    offset = 0
    if not na(htf_prev_start) and not na(htf_prev_end)
        start_j = bar_index - htf_prev_end
        end_j   = bar_index - htf_prev_start
        for j = math.max(1, start_j) to math.min(490, end_j)
            match = isBull ? (math.abs(low[j] - lvl) < syminfo.mintick) : (math.abs(high[j] - lvl) < syminfo.mintick)
            if match
                offset := j
                break
    offset

f_alreadyTouched(isLowLevel, lvl, tf) =>
    int barsSinceStart = ta.barssince(ta.change(time(tf)))
    bool touched = false
    if not na(barsSinceStart) and barsSinceStart > 0
        int limit = math.min(barsSinceStart, 300)
        for i = 1 to limit
            if isLowLevel
                if low[i] <= lvl
                    touched := true
                    break
            else
                if high[i] >= lvl
                    touched := true
                    break
    touched

f_sizeOk(bkSize) =>
    pip = syminfo.mintick * (syminfo.type == "forex" ? 10.0 : 1.0)
    sz  = i_customSizeMode == "Pips" ? bkSize / pip : bkSize

    if i_sizeFilter == "All"
        true
    else if i_sizeFilter == "Custom"
        minOk = i_customMin == 0.0 or sz >= i_customMin
        maxOk = i_customMax == 0.0 or sz <= i_customMax
        minOk and maxOk
    else
        array.push(_bkSizeHistory, sz)
        if array.size(_bkSizeHistory) > 100
            array.shift(_bkSizeHistory)
        n = array.size(_bkSizeHistory)
        if n < 3
            true
        else
            sorted = array.copy(_bkSizeHistory)
            array.sort(sorted)
            p33 = array.get(sorted, math.round(n * 0.33))
            p66 = array.get(sorted, math.round(n * 0.66))
            isSmall  = sz < p33
            isMed    = sz >= p33 and sz < p66
            isLarge  = sz >= p66
            i_sizeFilter == "Small+Med+Large" ? true :
             i_sizeFilter == "Med+Large"      ? (isMed or isLarge) :
             i_sizeFilter == "Large"          ? isLarge : true

formatTimeframe(string tf) =>
    seconds = timeframe.in_seconds(tf)
    if seconds < 60
        str.tostring(seconds) + "S"
    else if (seconds / 60) < 60
        str.tostring(math.round(seconds/60)) + "M"
    else if (seconds/60/60) < 24
        str.tostring(math.round(seconds/60/60)) + "H"
    else if (seconds/60/60/24) < 7
        str.tostring(math.round(seconds/60/60/24)) + "D"
    else if (seconds/60/60/24/7) < 4
        str.tostring(math.round(seconds/60/60/24/7)) + "W"
    else if (seconds/60/60/24/30) < 12
        str.tostring(math.round(seconds/60/60/24/30)) + "M"
    else
        str.tostring(math.round(seconds/60/60/24/365)) + "Y"

f_checkSweep(srcEnable, htfPH, htfPL, swH, swL, liqType, tfLbl, tf) =>
    sweptBull = false
    sweptBear = false
    sweptLvl  = float(na)
    lbl       = ""
    if srcEnable
        useOhlc  = liqType == "OHLC + Swing" or liqType == "OHLC Only"
        useSwing = liqType == "OHLC + Swing" or liqType == "Swing Only"
        
        ohlcLowAlreadyTouched  = f_alreadyTouched(true,  htfPL, tf)
        ohlcHighAlreadyTouched = f_alreadyTouched(false, htfPH, tf)
        swingLowAlreadyTouched  = f_alreadyTouched(true,  swL,   tf)
        swingHighAlreadyTouched = f_alreadyTouched(false, swH,   tf)

        htf_curr_start = ta.valuewhen(ta.change(time(tf)), bar_index, 0)
        htf_prev_start = ta.valuewhen(ta.change(time(tf)), bar_index, 1)
        htf_prev_end   = htf_curr_start - 1

        sw_low_offset = f_findLevelOrigin(true, swL)
        swing_low_already_swept = false
        if not na(sw_low_offset) and sw_low_offset > 1
            for i = 1 to sw_low_offset - 1
                if low[i] <= swL
                    swing_low_already_swept := true
                    break

        sw_high_offset = f_findLevelOrigin(false, swH)
        swing_high_already_swept = false
        if not na(sw_high_offset) and sw_high_offset > 1
            for i = 1 to sw_high_offset - 1
                if high[i] >= swH
                    swing_high_already_swept := true
                    break

        if useOhlc and not na(htfPL) and not ohlcLowAlreadyTouched and low <= htfPL
            sweptBull := true
            sweptLvl  := htfPL
            lbl       := tfLbl + " OHLC"
                
        if useSwing and not na(swL) and not sweptBull and not swingLowAlreadyTouched and not swing_low_already_swept and low <= swL
            sweptBull := true
            sweptLvl  := swL
            lbl       := tfLbl + " Swing"
            
        if useOhlc and not na(htfPH) and not ohlcHighAlreadyTouched and high >= htfPH
            sweptBear := true
            sweptLvl  := htfPH
            lbl       := tfLbl + " OHLC"
            
        if useSwing and not na(swH) and not sweptBear and not swingHighAlreadyTouched and not swing_high_already_swept and high >= swH
            sweptBear := true
            sweptLvl  := swH
            lbl       := tfLbl + " Swing"
            
    [sweptBull, sweptBear, sweptLvl, lbl]

f_findBullBB(pivotOffset) =>
    bbTop = float(na)
    bbBot = float(na)
    bbLen = 0
    startIdx = na(pivotOffset) ? 1 : pivotOffset
    for i = startIdx to startIdx + 8
        if close[i] > open[i]
            bbTop := na(bbTop) ? high[i] : math.max(bbTop, high[i])
            bbBot := na(bbBot) ? low[i]  : math.min(bbBot, low[i])
            bbLen := (i - startIdx) + 1
        else
            if not na(bbTop)
                break
    if na(bbTop) and not na(pivotOffset)
        bbTop := high[pivotOffset]
        bbBot := low[pivotOffset]
        bbLen := 1
    [bbTop, bbBot, bbLen]

        for i = n - 1 to 0

            if s.state == 4 and i_discardInval
                continue
            if s.state >= 3
                continue

            isBull = s.isBull
            if s.state == 1
                retested = isBull ? (low <= s.zoneTop and low >= s.zoneBot) : (high >= s.zoneBot and high <= s.zoneTop)
                if retested
                    s.state := 2
                    newCol = isBull ? i_bullRetColor : i_bearRetColor
                    box.set_bgcolor(s.zoneBox, newCol)
                    box.set_border_color(s.zoneBox, color.new(newCol, 100))
                    box.set_text_color(s.zoneBox, color.black)

            if s.state == 1 or s.state == 2
                box.set_right(s.zoneBox,  bar_index + 6)
                if not na(s.bb50Line)
                    extendLine = s.isBull ? i_extendBB50Bull : i_extendBB50Bear
                    if extendLine
                        boxLeft = box.get_left(s.zoneBox)
                        line.set_x2(s.bb50Line, math.max(boxLeft, bar_index + 6))
                if array.size(s.fvgBoxes) > 0
                    for fi = 0 to array.size(s.fvgBoxes) - 1
                        curFvgBox = array.get(s.fvgBoxes, fi)
                        if (s.isBull and i_extendFVGBull) or (s.isBull == false and i_extendFVGBear)
                            box.set_right(curFvgBox, bar_index + 6)
                            if array.size(s.fvgCeLines) > fi
                                line.set_x2(array.get(s.fvgCeLines, fi), bar_index + 6)
                if not na(s.zoneLbl)
                    label.set_x(s.zoneLbl, bar_index + 8)

            if i_modelMode == "Breaker + Unicorn" and not s.isUnicorn
                bool fFound = false
                float[] fTops = na
                float[] fBots = na
                int[] fIdxs = na

                if s.isBull
                    [found, tops, bots, idxs] = f_findBullFvg(s.zoneTop, s.zoneBot, s.sweepBar)
                    fFound := found
                    fTops  := tops
                    fBots  := bots
                    fIdxs  := idxs
                else
                    [found, tops, bots, idxs] = f_findBearFvg(s.zoneTop, s.zoneBot, s.sweepBar)
                    fFound := found
                    fTops  := tops
                    fBots  := bots
                    fIdxs  := idxs

                    s.isUnicorn := true
                    box.set_text(s.zoneBox, s.isBull ? "+Unicorn" : "-Unicorn")
                    fvgCol = s.isBull ? i_bullFvgColor : i_bearFvgColor
                    for i = 0 to array.size(fTops) - 1

                        fb = array.get(fBots, i)

            if i_useBE and not s.isBEActive and (s.state == 1 or s.state == 2)
                beTriggerPrice = s.entryPrice + (s.mainTarget - s.entryPrice) * 0.5
                isTriggered = isBull ? (high >= beTriggerPrice) : (low <= beTriggerPrice)
                if isTriggered

                    s.alertedInvalWarn := true

                    if not na(s.bb50Line)
                        extendLine = s.isBull ? i_extendBB50Bull : i_extendBB50Bear
                        if extendLine
                            boxLeft = box.get_left(s.zoneBox)
                            line.set_x2(s.bb50Line, math.max(boxLeft, bar_index - 5))
                                   
                    if array.size(s.fvgBoxes) > 0

f_gc() =>
    while array.size(setups) > i_history
        oldest = array.shift(setups)
        box.delete(oldest.zoneBox)
        if array.size(oldest.fvgBoxes) > 0
            for fi = 0 to array.size(oldest.fvgBoxes) - 1
                box.delete(array.get(oldest.fvgBoxes, fi))
        line.delete(oldest.sweepLine)
        label.delete(oldest.sweepLbl)
        label.delete(oldest.zoneLbl)
        line.delete(oldest.bb50Line)
        if array.size(oldest.fvgCeLines) > 0
            for cel in oldest.fvgCeLines
                line.delete(cel)
        tln = array.size(oldest.targetLines)
        if tln > 0
            for ti = 0 to tln - 1
                line.delete(array.get(oldest.targetLines, ti))
        tlabn = array.size(oldest.targetLabels)
        if tlabn > 0
            for ti = 0 to tlabn - 1
                label.delete(array.get(oldest.targetLabels, ti))
        line.delete(oldest.beTriggerLine)
        label.delete(oldest.beTriggerLabel)

    if i_discardInval
        n = array.size(setups)
        if n > 0
            i = n - 1
            while i >= 0
                s = array.get(setups, i)
                if s.state == 4 or s.state == 5
                    box.delete(s.zoneBox)
                    if array.size(s.fvgBoxes) > 0
                        for fi = 0 to array.size(s.fvgBoxes) - 1
                            box.delete(array.get(s.fvgBoxes, fi))
                    line.delete(s.sweepLine)
                    label.delete(s.sweepLbl)
                    label.delete(s.zoneLbl)
                    line.delete(s.bb50Line)
                    if array.size(s.fvgCeLines) > 0
                        for cel in s.fvgCeLines
                            line.delete(cel)
                    tln = array.size(s.targetLines)
                    if tln > 0

        int losses = 0
        int bes    = 0
        
        for i = 0 to array.size(stat_years) - 1
            if array.get(stat_years, i) == y and array.get(stat_months, i) == m
                total  += 1
                outcome = array.get(stat_outcomes, i)
                if outcome == 1
                    wins += 1
                else if outcome == 0
                    losses += 1
                else if outcome == 2
                    bes += 1
                    
        winRate = (wins + losses) > 0 ? (wins * 100.0) / (wins + losses) : 0.0
        
        cellBg = r % 2 == 0 ? color.new(chart.bg_color, 10) : color.new(chart.bg_color, 25)
        
        table.cell(statsTbl, 0, r + 1, str.tostring(m) + "/" + str.tostring(y), text_color=chart.fg_color, bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
        table.cell(statsTbl, 1, r + 1, str.tostring(total),                     text_color=chart.fg_color, bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
        table.cell(statsTbl, 2, r + 1, str.tostring(wins),                      text_color=color.new(#26a69a, 0), bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
        table.cell(statsTbl, 3, r + 1, str.tostring(losses),                    text_color=color.new(#ef5350, 0), bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
        table.cell(statsTbl, 4, r + 1, str.tostring(bes),                       text_color=color.new(#ff9800, 0), bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
        table.cell(statsTbl, 5, r + 1, str.tostring(math.round(winRate)) + "%", text_color=chart.fg_color, bgcolor=cellBg, text_size=size.small, text_font_family=font.family_monospace)
