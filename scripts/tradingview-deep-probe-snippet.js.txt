      i,
      tag: e.tagName,
      text: (e.innerText || e.textContent || '').trim().replace(/\s+/g,' ').slice(0,220),
      aria: e.getAttribute('aria-label') || '',

      type: e.getAttribute('type') || '',

    })).filter(x => /deep|backtest|date range|update report|reset to chart|strategy report|properties|start date|end date/i.test([x.text,x.aria,x.title,x.dataName].join(' ')));
    return {bodyText:(document.body.innerText||'').slice(0,70000),hits};

    page.locator('[title*="Deep Backtesting" i]').first(),

    page.getByText(/Deep Backtesting/i,{exact:false}).first()

      })).filter(x=>/deep|backtest|date|start|end|update report|reset to chart/i.test([x.text,x.aria,x.title,x.dataName,x.placeholder].join(' '))).slice(0,250)
    }));
    fs.writeFileSync('/tmp/tv-deep-after-click.json',JSON.stringify(after,null,2));
