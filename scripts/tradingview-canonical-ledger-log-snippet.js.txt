  // Open Pine Logs from the editor More menu and capture all virtualized parity lines.
  const __sym=process.env.TV_SYMBOL||'BNBUSDT';
  await p.waitForTimeout(2500);
  const editorControls=await dlg.locator('button,[role="button"],[aria-label],[title],[data-name]').evaluateAll(es=>es.map((e,i)=>({i,text:(e.innerText||e.textContent||'').trim().replace(/\s+/g,' ').slice(0,140),aria:e.getAttribute('aria-label')||'',title:e.getAttribute('title')||'',dataName:e.getAttribute('data-name')||'',cls:String(e.className||'').slice(0,180)})));
  fs.writeFileSync(`/tmp/tv-pine-editor-controls-${__sym}.json`,JSON.stringify(editorControls,null,2));
  let more=dlg.locator('button[aria-label*="More" i],button[title*="More" i],[role="button"][aria-label*="More" i],[data-name*="more" i],[data-name*="menu" i]').first();
  if(await more.count()===0){
    const btns=dlg.locator('button'); const n=await btns.count();
    if(n>0) more=btns.nth(n-1);
  }
  log('TV_PINE_MORE_CONTROL',await more.count()>0?'PRESENT':'ABSENT');
  if(await more.count()>0){await more.click({force:true}).catch(()=>{});await p.waitForTimeout(1200)}
  const menuDump=await p.locator('[role="menuitem"], [role="menu"] *, [data-name*="menu" i] *').evaluateAll(es=>es.map(e=>({text:(e.innerText||e.textContent||'').trim().replace(/\s+/g,' '),role:e.getAttribute('role')||'',aria:e.getAttribute('aria-label')||'',cls:String(e.className||'').slice(0,180)})).filter(x=>x.text).slice(0,250)).catch(()=>[]);
  fs.writeFileSync(`/tmp/tv-pine-more-menu-${__sym}.json`,JSON.stringify(menuDump,null,2));
  let logItem=p.getByRole('menuitem',{name:/^Pine logs(?:…|\.\.\.)?$/i}).first();
  if(await logItem.count()===0) logItem=p.locator('[role="menu"] [role="menuitem"]:has-text("Pine logs")').first();
  log('TV_PINE_LOG_MENU_ITEM',await logItem.count()>0?'PRESENT':'ABSENT');
  if(await logItem.count()>0){await logItem.click({force:true});await p.waitForTimeout(4500)}

  const collected=await p.evaluate(async()=>{
    const sleep=ms=>new Promise(r=>setTimeout(r,ms));
    const out=new Set();
    const grab=()=>{
      for(const line of (document.body.innerText||'').split('\n')){
        const s=line.trim();
        if(/WRP[123]\|/.test(s)) out.add(s);
      }
    };
    grab();
    const els=[...document.querySelectorAll('*')].filter(e=>{
      const r=e.getBoundingClientRect();
      return e.scrollHeight>e.clientHeight+30 && r.width>200 && r.height>80;
    }).map(e=>{
      const t=(e.innerText||e.textContent||'');
      let score=0;
      if(/WRP[123]\|/.test(t)) score+=100;
      if(/Pine logs/i.test(t)) score+=20;
      if(/log\.info|Info/i.test(t)) score+=5;
      score+=Math.min(20,Math.round((e.scrollHeight-e.clientHeight)/500));
      return {e,score,max:e.scrollHeight-e.clientHeight};
    }).sort((a,b)=>b.score-a.score).slice(0,12);
    for(const x of els){
      const positions=[];
      for(let f=0;f<=1.0001;f+=0.05)positions.push(Math.round(x.max*Math.min(1,f)));
      positions.push(x.max);
      for(const pos of [...new Set(positions)]){
        x.e.scrollTop=pos;
        x.e.dispatchEvent(new Event('scroll',{bubbles:true}));
        await sleep(140);
        grab();
      }
      for(const pos of [...new Set(positions)].reverse()){
        x.e.scrollTop=pos;
        x.e.dispatchEvent(new Event('scroll',{bubbles:true}));
        await sleep(100);
        grab();
      }
    }
    return {lines:[...out],scrollables:els.map(x=>({score:x.score,max:x.max,tag:x.e.tagName,cls:String(x.e.className||'').slice(0,180)}))};
  });
  const logBody=await p.locator('body').innerText().catch(()=>'');
  fs.writeFileSync(`/tmp/tv-pine-log-body-${__sym}.txt`,logBody+'\n');
  fs.writeFileSync(`/tmp/tv-pine-log-scrollables-${__sym}.json`,JSON.stringify(collected.scrollables,null,2));
  const parityLines=[...new Set(collected.lines)].sort((a,b)=>{
    const an=Number((a.match(/\|n=(\d+)/)||[])[1]||0), bn=Number((b.match(/\|n=(\d+)/)||[])[1]||0);
    if(an!==bn)return an-bn;
    const at=Number((a.match(/WRP([123])\|/)||[])[1]||0), bt=Number((b.match(/WRP([123])\|/)||[])[1]||0);
    return at-bt;
  });
  fs.writeFileSync(`/tmp/tv-canonical-ledger-${__sym}.txt`,parityLines.join('\n')+(parityLines.length?'\n':''));
  log('TV_PINE_PARITY_LOG_LINES',String(parityLines.length));
  await p.screenshot({path:`/tmp/tv-pine-logs-${__sym}.png`,fullPage:true}).catch(()=>{});
